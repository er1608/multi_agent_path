cmake_minimum_required(VERSION 3.16)
project(mapf_tv4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(mapf_tv4 src/validator.cpp src/joint_state_oracle.cpp src/analysis.cpp src/bypass.cpp src/io.cpp)
target_include_directories(mapf_tv4 PUBLIC include)
add_executable(tv4_cli src/main.cpp)
target_link_libraries(tv4_cli PRIVATE mapf_tv4)
add_executable(tv4_demo examples/demo.cpp)
target_link_libraries(tv4_demo PRIVATE mapf_tv4)
enable_testing()
add_executable(tv4_tests tests/tv4_tests.cpp)
target_link_libraries(tv4_tests PRIVATE mapf_tv4)
add_test(NAME tv4_tests COMMAND tv4_tests)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME python_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
  set_tests_properties(python_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
add_executable(oracle_tests tests/oracle_tests.cpp)
target_link_libraries(oracle_tests PRIVATE mapf_tv4)
add_test(NAME oracle_tests COMMAND oracle_tests)
